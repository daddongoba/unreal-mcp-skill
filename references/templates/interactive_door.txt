--- Components (BP_Door) ---
Root: DoorPanel(cube) or SceneRoot; child: TouchBox = ActorTools.add_component(
  owner=CDO, component_type="/Script/Engine.BoxComponent", name="TouchBox")
  boxExtent (~2x panel half-extents, z spanning contact zone), bGenerateOverlapEvents=true,
  bodyInstance.collisionProfileName="Trigger"  (verify readback: QueryOnly)
vars: TargetYaw(float), CurrentYaw(float), bOpening(bool→nodes are GetOpening/SetOpening #27)

--- DSL: overlap opens ---
(event AddEvent|Collision|EventActorBeginOverlap
  (Variables|Default|SetTargetYaw 90.0)
  (Variables|Default|SetOpening true)
  (Development|PrintString :InString "DOOR TRIGGERED"))

--- DSL: Tick lerp (#38: RInterpTo wants ROTATORS) ---
(event EventTick (DeltaSeconds)
  (bind opening (Variables|Default|GetOpening))
  (if opening
    (bind cur (Variables|Default|GetCurrentYaw))
    (bind tgt (Variables|Default|GetTargetYaw))
    (bind curRot (Math|Rotator|MakeRotator :Pitch 0.0 :Yaw cur :Roll 0.0))
    (bind tgtRot (Math|Rotator|MakeRotator :Pitch 0.0 :Yaw tgt :Roll 0.0))
    (bind newRot (Math|Interpolation|RInterpTo :Current curRot :Target tgtRot :DeltaTime DeltaSeconds :InterpSpeed 5.0))
    (Transformation|SetActorRotation :NewRotation newRot)
    (Variables|Default|SetCurrentYaw (.yaw newRot))))
